Quick Setup & Maven Commands

# 1. Clone repository
git clone https://github.com/tahniatfarhan/bank-management-system-java.git
cd bank-management-system-java

# 2. Configure Environment Variables
export DB_HOST=localhost
export DB_PORT=3306
export DB_NAME=bank_db
export DB_USER=root
export DB_PASSWORD=your_password

# 3. Build & Run JUnit 5 Tests
mvn clean test

# 4. Launch Desktop GUI Application
mvn exec:java -Dexec.mainClass="com.bank.Main"
